    基础指令¶

    x86 汇编核心指令集

    包括逻辑运算、移位、字符串操作和控制转移指令。

    🔣 逻辑运算指令¶

    AND, OR, XOR, NOT, TEST¶

    指令 功能 示例
    AND 按位与 and ax, bx
    OR 按位或 or ax, bx
    XOR 按位异或 xor ax, ax (清零)
    NOT 按位取反 not ax
    TEST 按位与，只影响标志位 test ax, 8000h

    TEST 指令

    test 和 and 的关系相当于 cmp 和 sub 的关系： 执行运算但不保存结果，只设置标志位。

    mov ax, 9234h
test ax, 8000h      ; ZF = 0, AX 不变
jnz msb_is_one      ; 最高位是 1

    判断寄存器是否为零¶

    ; 以下方法等价:
test cl, cl
or cl, cl
and cl, cl
cmp cl, 0

    ↔️ 移位指令¶

    逻辑移位 (无符号数)¶

    指令 功能
    SHL 逻辑左移 (右边补 0)
    SHR 逻辑右移 (左边补 0)

    算术移位 (有符号数)¶

    指令 功能
    SAL 算术左移 (= SHL)
    SAR 算术右移 (左边补符号位) 
    mov ah, 11111100B   ; AH = -4
sar ah, 2           ; AH = 11111111B = -1

    循环移位¶

    指令 功能
    ROL 循环左移
    ROR 循环右移
    RCL 带进位循环左移
    RCR 带进位循环右移 
    mov ah, 0B6h        ; 1011 0110
rol ah, 1           ; AH = 0110 1101, CF = 1

    CF 的作用

    所有移位指令最后移出的位都会保存在 CF 中。

    输出 16 位二进制 
    mov cx, 16
next:
    shl ax, 1       ; 最高位移入 CF
    mov dl, '0'
    adc dl, 0       ; CF=1 时 dl='1'
    push ax
    mov ah, 2
    int 21h
    pop ax
    dec cx
    jnz next

    📜 字符串操作指令¶

    准备工作¶

    寄存器 作用
    DS:SI 源字符串地址
    ES:DI 目标字符串地址
    CX 操作次数
    DF 方向 (0=正向, 1=反向)

    MOVSB/MOVSW/MOVSD - 字符串传送¶

    cld                 ; DF = 0, 正方向
rep movsb           ; 复制 CX 个字节

    rep movsb 的操作过程

    again:
    if (cx == 0) goto done
    es:[di] = ds:[si]
    if (df == 0) { si++; di++; }
    else { si--; di--; }
    cx--
    goto again
done:
    复制字符串 
    mov ax, 1000h
mov ds, ax
mov si, 0           ; 源: 1000:0000
mov ax, 2000h
mov es, ax
mov di, 0           ; 目标: 2000:0000
mov cx, 4           ; 复制 4 字节
cld                 ; 正方向
rep movsb

    CMPSB - 字符串比较¶

    repe cmpsb          ; 相等则继续比较 (repeat if equal)
repne cmpsb         ; 不等则继续比较 (repeat if not equal)

    SCASB - 字符串扫描¶

    在 ES:DI 指向的字符串中查找 AL 中的字符。

    mov al, 0           ; 查找 '\0'
mov cx, 0FFFFh      ; 最多查找 65535 次
cld
repne scasb         ; 不等则继续
not cx              ; cx = strlen
dec cx

    STOSB - 存储字符串¶

    rep stosb           ; 将 AL 填充到 ES:DI，共 CX 次
; 类似 memset(es:di, al, cx)

    LODSB - 加载字符串¶

    lodsb               ; AL = DS:[SI], SI++
    过滤字符 
    ; 过滤字符串中的 '#'
; DS:SI -> "##AB#12#XY"
; ES:DI -> 输出缓冲区
cld
again:
    lodsb           ; AL = DS:[SI++]
    cmp al, '#'
    je next
    stosb           ; ES:[DI++] = AL
next:
    dec cx
    jnz again
; 结果: ES:DI -> "AB12XY"

    🔀 控制转移指令¶

    JMP - 跳转¶

    类型 格式 范围
    短跳 jmp short target -128 ~ +127
    近跳 jmp near ptr target 段内任意
    远跳 jmp far ptr target 跨段

    编译器会自动选择跳转类型

    一般直接写 jmp target 即可。

    条件跳转 (短跳)¶

    指令 条件 说明
    jz / je ZF = 1 零/相等
    jnz / jne ZF = 0 非零/不等
    jc / jb CF = 1 进位/低于 (无符号)
    jnc / jae CF = 0 无进位/高于等于
    js SF = 1 负数
    jns SF = 0 非负
    jo OF = 1 溢出
    jno OF = 0 无溢出
    ja CF=0 且 ZF=0 高于 (无符号)
    jg SF=OF 且 ZF=0 大于 (有符号)
    jl SF≠OF 小于 (有符号)

    LOOP - 循环¶

    mov cx, 10
next:
    ; ... 循环体 ...
    loop next       ; cx--, if(cx≠0) goto next

    LOOP 先减后判断

    如果初始 CX = 0，会循环 65536 次！ 使用 jcxz done 可以避免。

    CALL 和 RET¶

    ; 近调用
call func           ; push ip, jmp func
ret                 ; pop ip

; 远调用
call far ptr func   ; push cs, push ip, jmp far func
retf                ; pop ip, pop cs

    📦 数据传送指令¶

    MOV¶

    限制

    两个操作数不能同时为内存变量

    ; 错误:
mov byte ptr ds:[bx], byte ptr es:[di]

; 正确:
mov al, es:[di]
mov ds:[bx], al

    XCHG - 交换¶

    xchg ax, bx         ; 交换寄存器
xchg ax, ds:[bx]    ; 交换寄存器和内存
; 不允许交换两个内存变量

    PUSH 和 POP¶

    push eax            ; 32位: esp-=4, [esp]=eax
pop ebx             ; 32位: ebx=[esp], esp+=4
