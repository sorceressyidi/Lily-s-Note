    寄存器详解¶

    寄存器是 CPU 内部的高速存储单元

    理解寄存器是学习汇编语言的基础。

    📌 堆栈初始化¶

    未定义堆栈时的默认值

    如果源程序中没有定义堆栈段：

    • SS = 首段的段地址 (如 1000h)
    • SP = 0
    • SP - 2 = FFFE (达到最大堆栈)

    注意：如果 push 太多，可能进入 code 段末尾，导致堆栈溢出！

    🚩 标志寄存器 (FL)¶

    FL 共 16 位，使用其中 9 位：6 个状态标志 + 3 个控制标志

    Bit:  11   10    9    8    7    6    4    2    0
Flag:  OF   DF   IF   TF   SF   ZF   AF   PF   CF

    状态标志¶

    CF (Carry Flag) - 进位标志¶

    用于无符号数运算

    当加法产生进位或减法产生借位时，CF = 1

    mov ah, 0FFh
add ah, 1       ; AH = 0, CF = 1 (产生进位)
add ah, 2       ; AH = 2, CF = 0
sub ah, 3       ; AH = 0FFh, CF = 1 (产生借位)

    移位指令也会影响 CF：最后移出的位保存在 CF 中

    mov cl, 2
shr ah, cl      ; 右移两位，CF = 最后移出的位

    相关指令：

    指令 功能
    jc 有进位则跳转
    jnc 无进位则跳转
    adc ax, bx ax = ax + bx + CF
    clc CF = 0
    stc CF = 1
    示例：输出二进制表示 
    data segment
    abc dw 32767
data ends
code segment
assume cs:code, ds:data
main:
    mov ax, data
    mov ds, ax
    mov ax, [abc]
    mov cx, 16
again:
    shl ax, 1
    jc is_one
is_zero:
    mov dl, '0'
    jmp output
is_one:
    mov dl, '1'
output:
    push ax
    mov ah, 2
    int 21h
    pop ax
    sub cx, 1
    jnz again
    mov ah, 4Ch
    int 21h
code ends
end main

    ZF (Zero Flag) - 零标志¶

    运算结果为零时 ZF = 1

    sub ax, ax      ; AX = 0, ZF = 1
add ax, 1       ; AX = 1, ZF = 0
add ax, 0FFFFh  ; AX = 0, ZF = 1, CF = 1

    相关指令：

    指令 等价 功能
    jz je ZF = 1 时跳转
    jnz jne ZF = 0 时跳转 
    cmp ax, ax      ; ax - ax，差值不保留，仅设置 ZF
                ; 判断相等 ↔ 判断 ZF = 1

    SF (Sign Flag) - 符号标志¶

    等于运算结果的最高位

    mov ah, 7Fh
add ah, 1       ; AH = 80h = 1000 0000B, SF = 1 (负数)
sub ah, 1       ; AH = 7Fh = 0111 1111B, SF = 0 (正数)

    相关指令：js (SF=1 跳转), jns (SF=0 跳转)

    OF (Overflow Flag) - 溢出标志¶

    用于有符号数运算

    当有符号数运算结果超出表示范围时 OF = 1

    mov ah, 7Fh
add ah, 1       ; AH = 80h, OF = 1 (正数变负数，溢出!)

mov ah, 80h
add ah, 0FFh    ; AH = 7Fh, OF = 1 (负数变正数，溢出!)

    相关指令：jo (OF=1 跳转), jno (OF=0 跳转)

    PF (Parity Flag) - 奇偶标志¶

    只统计低 8 位中 1 的个数

    偶数个 1 时 PF = 1，奇数个 1 时 PF = 0

    mov ah, 4
add ah, 1       ; AH = 0000 0101B, PF = 1 (2个1，偶数)

    相关指令：

    指令 等价 功能
    jp jpe PF = 1 时跳转 (parity even)
    jnp jpo PF = 0 时跳转 (parity odd)
    应用：奇偶校验 
    ; 接收 8 位值在 AL 中，采用偶校验验证
or al, al
jnp error       ; if(PF == 0) goto error
good:
    ...
error:
    ...

    AF (Auxiliary Flag) - 辅助进位标志¶

    低 4 位向高 4 位产生进位或借位时 AF = 1

    mov ah, 1Fh     ; 0001 1111
add ah, 1       ; ah = 20h, AF = 1

    AF 与 BCD 码运算相关

    控制标志¶

    DF (Direction Flag) - 方向标志¶

    控制字符串操作方向

    • DF = 0：正方向 (低地址 → 高地址)
    • DF = 1：反方向 (高地址 → 低地址)

    设置指令：

    指令 功能
    cld DF = 0 (正方向)
    std DF = 1 (反方向)

    复制方向选择：

    • 若 源地址 > 目标地址 → 正方向
    • 若 源地址 < 目标地址 → 反方向
    字符串复制示例 
    data segment
    s db "ABC123xyz", 0
    t db 10 dup(0)
data ends
code segment
assume cs:code, ds:data
main:
    mov ax, data
    mov ds, ax
    mov es, ax
    mov si, offset s + 9
    mov di, offset t + 9
    mov cx, 10
    std                 ; 反方向
    rep movsb           ; memcpy(es:di, ds:si, cx)
done:
    mov ah, 4Ch
    int 21h
code ends
end main

    IF (Interrupt Flag) - 中断标志¶

    控制硬件中断

    • IF = 1：允许中断
    • IF = 0：禁止中断
    指令 功能
    cli IF = 0 (禁止硬件中断)
    sti IF = 1 (允许硬件中断)

    保护关键代码：

    cli             ; 禁止硬件中断
; ... 重要代码 (不会被打断) ...
sti             ; 允许硬件中断

    中断向量表：

    • INT n 对应的中断向量存储在 0:n*4
    • 范围：[00, FF]，共 256 个，占用 0:0 ~ 0:3FF (400h 字节)

    TF (Trap Flag) - 单步标志¶

    调试用

    TF = 1 时，CPU 进入单步模式，每执行完一条指令后自动执行 INT 1h

    应用：反调试技术
